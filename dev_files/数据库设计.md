## 2. 建表语句（PostgreSQL + pgvector）

```sql
-- 启用 pgvector 扩展
CREATE EXTENSION IF NOT EXISTS vector;

-- ============================================
-- 1. 用户权限体系
-- ============================================

CREATE TABLE users (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    name VARCHAR(100) NOT NULL,
    email VARCHAR(255) UNIQUE NOT NULL,
    password_hash VARCHAR(255) NOT NULL,
    status VARCHAR(20) NOT NULL DEFAULT 'active' CHECK (status IN ('active', 'inactive', 'locked')),
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

CREATE INDEX idx_users_email ON users(email);
CREATE INDEX idx_users_status ON users(status);

CREATE TABLE roles (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    name VARCHAR(50) UNIQUE NOT NULL,
    description TEXT,
    permissions JSONB NOT NULL DEFAULT '[]',
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

CREATE TABLE user_roles (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    role_id UUID NOT NULL REFERENCES roles(id) ON DELETE CASCADE,
    scope_type VARCHAR(20) DEFAULT 'global' CHECK (scope_type IN ('global', 'process', 'instance')),
    scope_id UUID,
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    UNIQUE(user_id, role_id, scope_type, scope_id)
);

CREATE INDEX idx_user_roles_user ON user_roles(user_id);
CREATE INDEX idx_user_roles_role ON user_roles(role_id);

-- 初始化默认角色
INSERT INTO roles (name, description, permissions) VALUES
('designer', '流程设计者', '["process:create", "process:edit", "process:publish", "template:use"]'),
('employee', '执行员工', '["task:execute", "task:submit", "instance:view:self"]'),
('manager', '管理者', '["task:approve", "instance:view:all", "report:view", "dashboard:view"]'),
('admin', '系统管理员', '["*"]');

-- ============================================
-- 2. 流程定义
-- ============================================

CREATE TABLE process_definitions (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    name VARCHAR(200) NOT NULL,
    version INTEGER NOT NULL DEFAULT 1,
    graph_json JSONB NOT NULL DEFAULT '{"nodes": [], "edges": []}',
    status VARCHAR(20) NOT NULL DEFAULT 'draft' CHECK (status IN ('draft', 'published', 'archived')),
    created_by UUID REFERENCES users(id),
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    UNIQUE(name, version)
);

CREATE INDEX idx_process_definitions_status ON process_definitions(status);
CREATE INDEX idx_process_definitions_created_by ON process_definitions(created_by);

CREATE TABLE node_definitions (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    process_id UUID NOT NULL REFERENCES process_definitions(id) ON DELETE CASCADE,
    node_name VARCHAR(200) NOT NULL,
    node_type VARCHAR(20) NOT NULL DEFAULT 'task' CHECK (node_type IN ('start', 'task', 'end')),
    input_spec JSONB NOT NULL DEFAULT '{}',
    action_spec JSONB NOT NULL DEFAULT '{}',
    output_spec JSONB NOT NULL DEFAULT '{}',
    ai_config JSONB NOT NULL DEFAULT '{}',
    progress_config JSONB NOT NULL DEFAULT '{}',
    sort_order INTEGER NOT NULL DEFAULT 0,
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

CREATE INDEX idx_node_definitions_process ON node_definitions(process_id);
CREATE INDEX idx_node_definitions_sort ON node_definitions(process_id, sort_order);

-- ============================================
-- 3. 流程实例与节点实例
-- ============================================

CREATE TABLE process_instances (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    definition_id UUID NOT NULL REFERENCES process_definitions(id),
    status VARCHAR(20) NOT NULL DEFAULT 'running' CHECK (status IN ('running', 'completed', 'terminated', 'paused')),
    planned_start_date DATE,
    actual_start_date TIMESTAMPTZ,
    current_node_id UUID REFERENCES node_instances(id),
    created_by UUID NOT NULL REFERENCES users(id),
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

CREATE INDEX idx_process_instances_status ON process_instances(status);
CREATE INDEX idx_process_instances_definition ON process_instances(definition_id);
CREATE INDEX idx_process_instances_created_by ON process_instances(created_by);

CREATE TABLE node_instances (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    instance_id UUID NOT NULL REFERENCES process_instances(id) ON DELETE CASCADE,
    definition_id UUID NOT NULL REFERENCES node_definitions(id),
    status VARCHAR(20) NOT NULL DEFAULT 'waiting' CHECK (status IN ('waiting', 'in_progress', 'pending_approval', 'completed', 'rejected')),
    planned_start_date DATE,
    planned_end_date DATE,
    actual_start_date TIMESTAMPTZ,
    actual_end_date TIMESTAMPTZ,
    percent_complete INTEGER NOT NULL DEFAULT 0 CHECK (percent_complete BETWEEN 0 AND 100),
    input_data JSONB DEFAULT '{}',
    output_data JSONB DEFAULT '{}',
    assignee_user_id UUID REFERENCES users(id),
    responsible_role_id UUID REFERENCES roles(id),
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

CREATE INDEX idx_node_instances_instance ON node_instances(instance_id);
CREATE INDEX idx_node_instances_status ON node_instances(status);
CREATE INDEX idx_node_instances_assignee ON node_instances(assignee_user_id);

CREATE TABLE node_instance_history (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    node_instance_id UUID NOT NULL REFERENCES node_instances(id) ON DELETE CASCADE,
    event_type VARCHAR(30) NOT NULL CHECK (event_type IN ('submit', 'ai_inspection_pass', 'ai_inspection_reject', 'approve', 'reject', 'timeout', 'reassign', 'terminate')),
    actor_user_id UUID REFERENCES users(id),
    from_status VARCHAR(20),
    to_status VARCHAR(20),
    details JSONB DEFAULT '{}',
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

CREATE INDEX idx_node_instance_history_node ON node_instance_history(node_instance_id);
CREATE INDEX idx_node_instance_history_created ON node_instance_history(created_at);

-- ============================================
-- 4. 任务（待办）
-- ============================================

CREATE TABLE tasks (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    node_instance_id UUID NOT NULL REFERENCES node_instances(id) ON DELETE CASCADE,
    assignee_user_id UUID NOT NULL REFERENCES users(id),
    type VARCHAR(20) NOT NULL CHECK (type IN ('execute', 'approve')),
    status VARCHAR(20) NOT NULL DEFAULT 'pending' CHECK (status IN ('pending', 'in_progress', 'completed', 'cancelled')),
    due_date TIMESTAMPTZ,
    completed_at TIMESTAMPTZ,
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

CREATE INDEX idx_tasks_assignee ON tasks(assignee_user_id);
CREATE INDEX idx_tasks_status ON tasks(status);
CREATE INDEX idx_tasks_type ON tasks(type);

-- ============================================
-- 5. 知识库
-- ============================================

CREATE TABLE knowledge_bases (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    name VARCHAR(200) NOT NULL,
    type VARCHAR(20) NOT NULL CHECK (type IN ('standard', 'expertise')),
    description TEXT,
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

CREATE INDEX idx_knowledge_bases_type ON knowledge_bases(type);

CREATE TABLE knowledge_base_documents (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    knowledge_base_id UUID NOT NULL REFERENCES knowledge_bases(id) ON DELETE CASCADE,
    file_name VARCHAR(500) NOT NULL,
    file_path VARCHAR(1000) NOT NULL,
    file_size BIGINT NOT NULL DEFAULT 0,
    mime_type VARCHAR(100),
    parse_status VARCHAR(20) NOT NULL DEFAULT 'pending' CHECK (parse_status IN ('pending', 'parsing', 'completed', 'failed')),
    vectorized_status VARCHAR(20) NOT NULL DEFAULT 'pending' CHECK (vectorized_status IN ('pending', 'processing', 'completed', 'failed')),
    chunk_count INTEGER DEFAULT 0,
    error_message TEXT,
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

CREATE INDEX idx_kb_docs_knowledge_base ON knowledge_base_documents(knowledge_base_id);
CREATE INDEX idx_kb_docs_status ON knowledge_base_documents(parse_status, vectorized_status);

-- 向量存储表（pgvector）
CREATE TABLE document_chunks (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    document_id UUID NOT NULL REFERENCES knowledge_base_documents(id) ON DELETE CASCADE,
    chunk_index INTEGER NOT NULL,
    content TEXT NOT NULL,
    embedding vector(1536),  -- OpenAI text-embedding-3-small 维度
    metadata JSONB DEFAULT '{}',
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

CREATE INDEX idx_document_chunks_document ON document_chunks(document_id);
CREATE INDEX idx_document_chunks_embedding ON document_chunks USING ivfflat (embedding vector_cosine_ops);

-- ============================================
-- 6. AI 相关
-- ============================================

CREATE TABLE ai_agent_logs (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    node_instance_id UUID REFERENCES node_instances(id),
    role VARCHAR(20) NOT NULL CHECK (role IN ('inspector', 'assistant')),
    input JSONB NOT NULL,
    output JSONB,
    latency_ms INTEGER,
    is_fallback BOOLEAN NOT NULL DEFAULT FALSE,
    error_message TEXT,
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

CREATE INDEX idx_ai_agent_logs_node ON ai_agent_logs(node_instance_id);
CREATE INDEX idx_ai_agent_logs_created ON ai_agent_logs(created_at);

CREATE TABLE ai_reports (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    node_instance_id UUID NOT NULL REFERENCES node_instances(id) ON DELETE CASCADE,
    report_type VARCHAR(20) NOT NULL CHECK (report_type IN ('inspector', 'assistant')),
    content JSONB NOT NULL,
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

CREATE INDEX idx_ai_reports_node ON ai_reports(node_instance_id);
CREATE INDEX idx_ai_reports_type ON ai_reports(report_type);

-- ============================================
-- 7. 审计与通知
-- ============================================

CREATE TABLE audit_logs (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID REFERENCES users(id),
    action VARCHAR(50) NOT NULL,
    resource_type VARCHAR(50) NOT NULL,
    resource_id UUID,
    details JSONB DEFAULT '{}',
    ip_address INET,
    user_agent TEXT,
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

CREATE INDEX idx_audit_logs_user ON audit_logs(user_id);
CREATE INDEX idx_audit_logs_resource ON audit_logs(resource_type, resource_id);
CREATE INDEX idx_audit_logs_created ON audit_logs(created_at);

CREATE TABLE notifications (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    type VARCHAR(30) NOT NULL,
    title VARCHAR(200) NOT NULL,
    content TEXT,
    channel VARCHAR(20) NOT NULL DEFAULT 'in_app' CHECK (channel IN ('in_app', 'email', 'both')),
    status VARCHAR(20) NOT NULL DEFAULT 'unread' CHECK (status IN ('unread', 'read')),
    related_resource_type VARCHAR(50),
    related_resource_id UUID,
    read_at TIMESTAMPTZ,
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

CREATE INDEX idx_notifications_user ON notifications(user_id);
CREATE INDEX idx_notifications_status ON notifications(status);
CREATE INDEX idx_notifications_created ON notifications(created_at);

-- ============================================
-- 8. 触发器：自动更新 updated_at
-- ============================================

CREATE OR REPLACE FUNCTION update_updated_at_column()
RETURNS TRIGGER AS $$
BEGIN
    NEW.updated_at = NOW();
    RETURN NEW;
END;
$$ language 'plpgsql';

CREATE TRIGGER update_users_updated_at BEFORE UPDATE ON users FOR EACH ROW EXECUTE FUNCTION update_updated_at_column();
CREATE TRIGGER update_process_definitions_updated_at BEFORE UPDATE ON process_definitions FOR EACH ROW EXECUTE FUNCTION update_updated_at_column();
CREATE TRIGGER update_node_definitions_updated_at BEFORE UPDATE ON node_definitions FOR EACH ROW EXECUTE FUNCTION update_updated_at_column();
CREATE TRIGGER update_process_instances_updated_at BEFORE UPDATE ON process_instances FOR EACH ROW EXECUTE FUNCTION update_updated_at_column();
CREATE TRIGGER update_node_instances_updated_at BEFORE UPDATE ON node_instances FOR EACH ROW EXECUTE FUNCTION update_updated_at_column();
CREATE TRIGGER update_tasks_updated_at BEFORE UPDATE ON tasks FOR EACH ROW EXECUTE FUNCTION update_updated_at_column();
CREATE TRIGGER update_knowledge_bases_updated_at BEFORE UPDATE ON knowledge_bases FOR EACH ROW EXECUTE FUNCTION update_updated_at_column();
CREATE TRIGGER update_kb_docs_updated_at BEFORE UPDATE ON knowledge_base_documents FOR EACH ROW EXECUTE FUNCTION update_updated_at_column();
```

## 3. 表设计说明

### 3.1 流程定义相关

| 表 | 用途 | 关键设计 |
|----|------|----------|
| `process_definitions` | 流程定义主表 | `graph_json` 存储节点连线关系；`version` + `status` 实现版本管理 |
| `node_definitions` | 节点定义 | `sort_order` 确定线性顺序；`input_spec/action_spec/output_spec` 为 JSONB 存储三阶段配置 |

### 3.2 流程实例相关

| 表 | 用途 | 关键设计 |
|----|------|----------|
| `process_instances` | 流程实例 | `current_node_id` 指向当前激活的节点实例；绑定 `definition_id` 保证版本不变 |
| `node_instances` | 节点实例 | `status` 字段驱动状态机；`assignee_user_id` + `responsible_role_id` 实现简化的人员模型 |
| `node_instance_history` | 节点执行历史 | 记录所有状态变更事件，支持审计和回溯 |
| `tasks` | 待办任务 | 员工/管理者的任务列表数据源；`type` 区分执行任务和审批任务 |

### 3.3 AI 相关

| 表 | 用途 | 关键设计 |
|----|------|----------|
| `ai_agent_logs` | AI 调用日志 | `is_fallback` 标记降级调用；`latency_ms` 监控性能 |
| `ai_reports` | AI 报告存储 | 独立表避免 node_instances 过大；JSONB 存储结构化报告 |

### 3.4 知识库相关

| 表 | 用途 | 关键设计 |
|----|------|----------|
| `knowledge_bases` | 知识库主表 | `type` 区分标准/规范 vs 专家技能 |
| `knowledge_base_documents` | 文档管理 | `parse_status` + `vectorized_status` 跟踪处理状态 |
| `document_chunks` | 文档分块向量 | `embedding` 使用 pgvector 的 `vector(1536)` 类型；IVFFlat 索引加速相似度检索 |

## 4. 索引策略

### 4.1 高频查询索引

```sql
-- 员工任务列表（按截止时间排序）
CREATE INDEX idx_tasks_assignee_due ON tasks(assignee_user_id, due_date) WHERE status = 'pending';

-- 流程实例列表（按状态筛选）
CREATE INDEX idx_process_instances_status_created ON process_instances(status, created_at);

-- 节点实例按流程实例查询
CREATE INDEX idx_node_instances_instance_status ON node_instances(instance_id, status);

-- 知识库文档检索（向量相似度）
CREATE INDEX idx_document_chunks_embedding ON document_chunks USING ivfflat (embedding vector_cosine_ops) WITH (lists = 100);
```

### 4.2 分区策略（MVP 暂不实现）

> 注：MVP 阶段数据量较小，暂不实施分区。V1.2 后若 `audit_logs` 和 `ai_agent_logs` 数据量超过 1000 万行，考虑按 `created_at` 月分区。

## 5. 数据保留策略

| 数据类型 | 保留期限 | 清理方式 |
|----------|----------|----------|
| 审计日志 (audit_logs) | 6 个月 | 定时任务删除 |
| AI 调用日志 (ai_agent_logs) | 3 个月 | 定时任务删除 |
| 通知 (notifications) | 已读 30 天，未读 90 天 | 定时任务删除 |
| 流程实例 | 永久 | 归档（V1.2 实现） |
| 流程定义 | 永久 | 软删除（status = archived） |
