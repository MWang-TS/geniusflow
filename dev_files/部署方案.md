## 2. Docker Compose 配置

```yaml
# docker-compose.yml
version: '3.8'

services:
  # 前端静态资源
  web:
    build:
      context: ./web
      dockerfile: Dockerfile
    container_name: geniusflow-web
    restart: unless-stopped
    networks:
      - geniusflow-network

  # 主 API 服务
  api:
    build:
      context: ./api
      dockerfile: Dockerfile
    container_name: geniusflow-api
    restart: unless-stopped
    environment:
      - NODE_ENV=production
      - DATABASE_URL=postgresql://postgres:password@postgres:5432/geniusflow
      - REDIS_URL=redis://redis:6379
      - AI_SERVICE_URL=http://ai-service:5000
      - JWT_SECRET=${JWT_SECRET}
      - JWT_EXPIRES_IN=2h
      - REFRESH_TOKEN_EXPIRES_IN=7d
    depends_on:
      postgres:
        condition: service_healthy
      redis:
        condition: service_healthy
    networks:
      - geniusflow-network
    volumes:
      - ./uploads:/app/uploads

  # AI 服务
  ai-service:
    build:
      context: ./ai-service
      dockerfile: Dockerfile
    container_name: geniusflow-ai
    restart: unless-stopped
    environment:
      - OPENAI_API_KEY=${OPENAI_API_KEY}
      - OPENAI_MODEL=gpt-4o-mini
      - EMBEDDING_MODEL=text-embedding-3-small
      - DATABASE_URL=postgresql://postgres:password@postgres:5432/geniusflow
      - REDIS_URL=redis://redis:6379
    depends_on:
      postgres:
        condition: service_healthy
      redis:
        condition: service_healthy
    networks:
      - geniusflow-network

  # PostgreSQL 数据库
  postgres:
    image: ankane/pgvector:v0.6.0
    container_name: geniusflow-postgres
    restart: unless-stopped
    environment:
      - POSTGRES_USER=postgres
      - POSTGRES_PASSWORD=${DB_PASSWORD}
      - POSTGRES_DB=geniusflow
    volumes:
      - postgres-data:/var/lib/postgresql/data
      - ./init.sql:/docker-entrypoint-initdb.d/init.sql
    networks:
      - geniusflow-network
    healthcheck:
      test: ["CMD-SHELL", "pg_isready -U postgres"]
      interval: 10s
      timeout: 5s
      retries: 5

  # Redis 缓存 + 队列
  redis:
    image: redis:7-alpine
    container_name: geniusflow-redis
    restart: unless-stopped
    volumes:
      - redis-data:/data
    networks:
      - geniusflow-network
    healthcheck:
      test: ["CMD", "redis-cli", "ping"]
      interval: 10s
      timeout: 5s
      retries: 5

  # Nginx 反向代理
  nginx:
    image: nginx:alpine
    container_name: geniusflow-nginx
    restart: unless-stopped
    ports:
      - "80:80"
      - "443:443"
    volumes:
      - ./nginx.conf:/etc/nginx/nginx.conf
      - ./ssl:/etc/nginx/ssl
    depends_on:
      - web
      - api
      - ai-service
    networks:
      - geniusflow-network

  # MinIO 对象存储（可选）
  minio:
    image: minio/minio:latest
    container_name: geniusflow-minio
    restart: unless-stopped
    environment:
      - MINIO_ROOT_USER=${MINIO_ROOT_USER}
      - MINIO_ROOT_PASSWORD=${MINIO_ROOT_PASSWORD}
    volumes:
      - minio-data:/data
    command: server /data --console-address ":9001"
    networks:
      - geniusflow-network
    profiles:
      - with-minio

volumes:
  postgres-data:
  redis-data:
  minio-data:

networks:
  geniusflow-network:
    driver: bridge
```

## 3. Nginx 配置

```nginx
# nginx.conf
user nginx;
worker_processes auto;
error_log /var/log/nginx/error.log warn;
pid /var/run/nginx.pid;

events {
    worker_connections 1024;
}

http {
    include /etc/nginx/mime.types;
    default_type application/octet-stream;

    log_format main '$remote_addr - $remote_user [$time_local] "$request" '
                    '$status $body_bytes_sent "$http_referer" '
                    '"$http_user_agent" "$http_x_forwarded_for"';

    access_log /var/log/nginx/access.log main;

    sendfile on;
    tcp_nopush on;
    tcp_nodelay on;
    keepalive_timeout 65;
    types_hash_max_size 2048;

    # Gzip 压缩
    gzip on;
    gzip_vary on;
    gzip_min_length 1024;
    gzip_types text/plain text/css application/json application/javascript text/xml;

    # WebSocket 支持
    map $http_upgrade $connection_upgrade {
        default upgrade;
        '' close;
    }

    # 上游服务
    upstream web {
        server web:3000;
    }

    upstream api {
        server api:4000;
    }

    upstream ai_service {
        server ai-service:5000;
    }

    # HTTP → HTTPS 重定向
    server {
        listen 80;
        server_name _;
        return 301 https://$host$request_uri;
    }

    # HTTPS 主服务
    server {
        listen 443 ssl http2;
        server_name geniusflow.example.com;

        ssl_certificate /etc/nginx/ssl/cert.pem;
        ssl_certificate_key /etc/nginx/ssl/key.pem;
        ssl_protocols TLSv1.2 TLSv1.3;
        ssl_ciphers HIGH:!aNULL:!MD5;

        # 前端静态资源
        location / {
            proxy_pass http://web;
            proxy_set_header Host $host;
            proxy_set_header X-Real-IP $remote_addr;
            proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
            proxy_set_header X-Forwarded-Proto $scheme;
        }

        # API 接口
        location /api/ {
            proxy_pass http://api/;
            proxy_set_header Host $host;
            proxy_set_header X-Real-IP $remote_addr;
            proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
            proxy_set_header X-Forwarded-Proto $scheme;
            
            # 超时设置
            proxy_connect_timeout 30s;
            proxy_send_timeout 30s;
            proxy_read_timeout 30s;
        }

        # WebSocket
        location /ws {
            proxy_pass http://api;
            proxy_http_version 1.1;
            proxy_set_header Upgrade $http_upgrade;
            proxy_set_header Connection $connection_upgrade;
            proxy_set_header Host $host;
            proxy_read_timeout 86400s;
        }

        # AI 服务（内部调用，不对外暴露）
        location /ai/ {
            internal;
            proxy_pass http://ai_service/;
        }

        # 文件上传大小限制
        client_max_body_size 100M;
    }
}
```

## 5. 部署脚本

```bash
#!/bin/bash
# deploy.sh

set -e

echo "🚀 GeniusFlow 部署脚本"

# 1. 检查环境
if [ ! -f .env ]; then
    echo "❌ 错误：.env 文件不存在"
    exit 1
fi

# 2. 加载环境变量
export $(cat .env | xargs)

# 3. 构建镜像
echo "📦 构建 Docker 镜像..."
docker-compose build

# 4. 启动服务
echo "🟢 启动服务..."
docker-compose up -d

# 5. 等待数据库就绪
echo "⏳ 等待数据库就绪..."
sleep 10

# 6. 执行数据库迁移
echo "🗄️ 执行数据库迁移..."
docker-compose exec -T api npx prisma migrate deploy

# 7. 初始化数据
echo "🌱 初始化默认数据..."
docker-compose exec -T api node dist/scripts/seed.js

# 8. 健康检查
echo "🏥 健康检查..."
curl -f http://localhost/api/health || exit 1

echo "✅ 部署完成！"
echo "🌐 访问地址：https://geniusflow.example.com"
```

## 6. 备份策略

```bash
#!/bin/bash
# backup.sh

BACKUP_DIR="/backup/geniusflow"
DATE=$(date +%Y%m%d_%H%M%S)

# 创建备份目录
mkdir -p $BACKUP_DIR

# 备份数据库
docker-compose exec -T postgres pg_dump -U postgres geniusflow | gzip > $BACKUP_DIR/db_$DATE.sql.gz

# 备份上传文件
tar czf $BACKUP_DIR/uploads_$DATE.tar.gz ./uploads

# 保留最近 7 天备份
find $BACKUP_DIR -name "*.gz" -mtime +7 -delete

echo "✅ 备份完成: $BACKUP_DIR/db_$DATE.sql.gz"
```

## 7. 监控与日志

### 7.1 日志收集

```yaml
# docker-compose.logging.yml
services:
  api:
    logging:
      driver: "json-file"
      options:
        max-size: "100m"
        max-file: "3"

  ai-service:
    logging:
      driver: "json-file"
      options:
        max-size: "100m"
        max-file: "3"
```

### 7.2 健康检查端点

| 服务 | 端点 | 说明 |
|------|------|------|
| API | GET /api/health | 数据库 + Redis 连接状态 |
| AI | GET /ai/health | OpenAI API 连接状态 |

### 7.3 关键监控指标

| 指标 | 采集方式 | 告警阈值 |
|------|----------|----------|
| API 响应时间 | Nginx access log | P99 > 500ms |
| 错误率 | Nginx access log | > 1% |
| 数据库连接数 | PostgreSQL stats | > 80% |
| Redis 内存使用 | Redis INFO | > 80% |
| 磁盘空间 | df | > 85% |
| AI 调用失败率 | AI service log | > 5% |

## 8. 升级流程

```bash

cd /opt/geniusflow
./backup.sh

# 2. 拉取最新代码

# 3. 重新构建

# 4. 滚动更新（零停机）

docker-compose up -d --no-deps --build ai-service

# 5. 执行迁移
docker-compose exec api npx prisma migrate deploy

curl -f http://localhost/api/health
```

## 10. 安全加固

| 措施 | 实现 |
|------|------|
| HTTPS | Nginx 配置 SSL 证书 |
| JWT 安全 | 2 小时过期 + refresh token |
| 密码加密 | bcrypt，salt rounds = 12 |
| SQL 注入 | Prisma ORM 参数化查询 |
| XSS 防护 | 前端转义 + CSP 头 |
| 文件上传 | 类型白名单 + 大小限制 |
| 操作审计 | audit_logs 记录所有敏感操作 |
| 失败锁定 | 5 次密码错误锁定 30 分钟 |
